import numpy as np

# --- Function 2: NumPy linalg Matrix Power ---
# Note: No @nb.jit decorator here
def fib_linalg_numpy(n):
    if n <= 0:
        return 0
    if n == 1:
        return 1
    Q = np.array([[1, 1], [1, 0]], dtype=np.uint64)
    # This is the line that cannot be compiled by Numba in nopython mode
    result_matrix = np.linalg.matrix_power(Q, n - 1)
    return result_matrix[0, 0]


@nb.njit(nb.uint64(nb.uint64), cache=True)
def fibonacci_iterative(n: int) -> int:
    """Calculate the nth Fibonacci number using an iterative approach."""
    if n < 0:
        raise ValueError("Fibonacci is not defined for negative numbers.")
    if n < 2:
        return n

    a, b = 0, 1
    for _ in range(n - 1):
        a, b = b, a + b
    return b

import numpy as np
import numba as nb


@nb.njit
def fib_matrix_procedural_fixed(n):
    """
    Calculates the n-th Fibonacci number with fixes for Numba compatibility.
    """
    if n <= 0:
        return 0
    if n == 1:
        return 1

    exponent = n - 1

    # FIX 1: Change dtype to np.float64 for BLAS compatibility
    base = np.array([[1, 1], [1, 0]], dtype=np.float64)
    result = np.array([[1, 0], [0, 1]], dtype=np.float64)

    while exponent > 0:
        if exponent % 2 == 1:
            # FIX 2: Use the @ operator for matrix multiplication
            result = result @ base

        # Use the @ operator here as well
        base = base @ base
        exponent //= 2

    # The result is a float, but it represents an integer.
    # We can safely cast it to an integer type for the final return.
    return np.uint64(result[0, 0])
